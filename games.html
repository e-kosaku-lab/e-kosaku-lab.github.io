サイコロと はんのうゲーム

さぼてんきけいのボードと画面で、遊べるプログラムを2つ作りました。さぼてんきくんが、登場します。ワイヤーは、1〜2本だけ。温度・湿度のセンサーは、使いません。

これは、つぎの一歩です。まず、いつもの「さぼてんきけい」が動いてから、試してね。ボードには、ここのプログラムを、書き込みなおします。（書き込むと、いつものプログラムは、いったん消えます。プログラムのページから、もう一度、書き込めば、もどります）

つなぎかた

画面（OLED）は、いつもの「さぼてんきけい」と同じ、4本のワイヤーで、つないだままです。センサーは、つないだままでも、はずしても、だいじょうぶです。

ワイヤーESP32のピンブレッドボードの穴つかいみち
ワイヤー1D148jサイコロ・はんのうゲーム（1P）
ワイヤー2D279jはんのうゲーム（2P）。1人で遊ぶときは、いらない

ワイヤーの、もう一方の先（金属の部分）は、どこにもつながず、あけておきます。ここを、さわります。くわしくは、タッチ版のページと、同じです。

USBケーブルは、はずしてからね。電源が入ったときは、ワイヤーの先に、さわらないでね。（画面に「さわらないでね」と出ている間に、さわっていない時の「ようす」を、おぼえます）

1. 電子サイコロ・おみくじ

ワイヤー1を、ちょんとさわると、サイコロが転がります。さぼてんきくんが、見まもります。

さわりかたおこること
ちょんサイコロを ふる。6が出ると、さぼてんきくんが「えへへ」。1が出ると「いたい!」
ながく（1.2秒）「サイコロ」と「おみくじ」を、きりかえる

かえてみよう

  • DICE_MAX：サイコロの面の数。6を、8や12にしてみよう。（7からは、数字で出ます）
  • OMIKUJI：おみくじの言葉。「だいきち」を、すきな言葉にしてみよう。
  • ROLL_MS：転がっている時間。
  • MARK_TERE など：さぼてんきくんの絵（0と1で、かきかえられます）。

U8G2_SSD1306_128X64_NONAME_F_HW_I2C display(U8G2_R0, U8X8_PIN_NONE);

const int MARK_SIZE = 16;
const int MARK_SCALE = 2;

struct TouchPad {
  int pin;
  float base;               // さわっていない時の 値（きじゅん）
  bool down;                // いま さわっているか
  int cnt;                  // ちょうし を みる ための かぞえ
  unsigned long pendMs;     // さわり はじめた とき
  unsigned long downMs;     // さわった ときこく
  bool longFired;
  bool pressedEdge;         // いま さわった
  bool releasedEdge;        // いま はなした
  bool longEdge;            // いま ながく さわった
  unsigned long heldMs;     // はなした とき の、さわっていた 長さ
};

// ESP32 は、さわると 値が 小さくなる
float measureBase(int pin) {
  long sum = 0;
  for (int i = 0; i < 16; i++) {
    sum += touchRead(pin);
    delay(4);
  }
  return sum / 16.0f;
}

void padUpdate(TouchPad &p, unsigned long now) {
  p.pressedEdge = false;
  p.releasedEdge = false;
  p.longEdge = false;
  float v = (float)touchRead(p.pin);
  bool raw = v < p.base * TOUCH_RATIO;
  if (raw == p.down) {
    p.cnt = 0;
    // さわっていない間は、きじゅんを、ゆっくり なおす
    if (!raw && v > p.base * 0.8f) p.base = p.base * 0.98f + v * 0.02f;
  } else {
    if (p.cnt == 0) p.pendMs = now;
    p.cnt++;
    if (p.cnt >= TOUCH_CONFIRM) {  // 3回つづけて 同じ → きまり（ちらつき よけ。かず は TOUCH_CONFIRM）
      p.down = raw;
      p.cnt = 0;
      if (raw) {
        p.pressedEdge = true;
        p.downMs = p.pendMs;
        p.longFired = false;
      } else {
        p.releasedEdge = true;
        p.heldMs = now - p.downMs;
      }
    }
  }
  if (p.down && !p.longFired && now - p.downMs >= LONG_MS) {
    p.longFired = true;
    p.longEdge = true;
  }
}

TouchPad pad1 = {14, 0, false, 0, 0, 0, false, false, false, false, 0};

// ---- 0と1の絵を、画面にかく（scale は 1ドットを 何倍で出すか）----
void drawMarkS(const char *const mark[], int x, int y, int scale) {
  for (int row = 0; row < MARK_SIZE; row++) {
    const char *line = mark[row];
    for (int col = 0; col < MARK_SIZE && line[col] != '\0'; col++) {
      if (line[col] == '1') {
        display.drawBox(x + col * scale, y + row * scale, scale, scale);
      }
    }
  }
}
void drawMark(const char *const mark[], int x, int y) {
  drawMarkS(mark, x, y, MARK_SCALE);
}

const int MODE_DICE = 0;
const int MODE_OMIKUJI = 1;
int mode = MODE_DICE;

bool rolling = false;
unsigned long rollStart = 0;
unsigned long nextChange = 0;
int shown = 1;    // いま 画面に出している 目（または おみくじの ばんごう）
int result = 0;   // さいごに きまった 目
bool hasResult = false;

int pickValue() {
  if (mode == MODE_DICE) return random(1, DICE_MAX + 1);
  return random(0, OMIKUJI_N);
}

void drawPip(int cx, int cy) {
  display.drawDisc(cx, cy, 4);
}

// サイコロの 面（左上が x,y。大きさ s）
void drawDiceFace(int v, int x, int y, int s) {
  display.drawRFrame(x, y, s, s, 6);
  if (v > 6) {
    char t[8];
    snprintf(t, sizeof(t), "%d", v);
    display.setFont(u8g2_font_logisoso28_tn);
    int w = display.getStrWidth(t);
    display.drawStr(x + (s - w) / 2, y + s / 2 + 14, t);
    display.setFont(u8g2_font_b12_t_japanese2);
    return;
  }
  int a = x + 12, b = x + s / 2, c = x + s - 12;
  int r1 = y + 12, r2 = y + s / 2, r3 = y + s - 12;
  switch (v) {
    case 1: drawPip(b, r2); break;
    case 2: drawPip(a, r1); drawPip(c, r3); break;
    case 3: drawPip(a, r1); drawPip(b, r2); drawPip(c, r3); break;
    case 4: drawPip(a, r1); drawPip(c, r1); drawPip(a, r3); drawPip(c, r3); break;
    case 5: drawPip(a, r1); drawPip(c, r1); drawPip(b, r2); drawPip(a, r3); drawPip(c, r3); break;
    default:
      drawPip(a, r1); drawPip(a, r2); drawPip(a, r3);
      drawPip(c, r1); drawPip(c, r2); drawPip(c, r3);
      break;
  }
}

// さぼてんきくんの 顔（ころがっている間は ぴょんぴょん）
void drawCactus(unsigned long now) {
  const char *const *m;
  int hop = 0;
  int frame = (now / ANIME_MS) % 2;
  if (rolling) {
    frame = (now / 120) % 2;
    hop = frame ? -3 : 0;
    m = frame ? MARK_KAITEKI2 : MARK_KAITEKI;
  } else if (hasResult) {
    bool great = (mode == MODE_DICE) ? (result == DICE_MAX) : (result == 0);
    bool bad = (mode == MODE_DICE) ? (result == 1 && DICE_MAX > 1) : (result == OMIKUJI_N - 1);
    if (great) m = frame ? MARK_TERE2 : MARK_TERE;
    else if (bad) m = frame ? MARK_ITAI2 : MARK_ITAI;
    else m = frame ? MARK_KAITEKI2 : MARK_KAITEKI;
  } else {
    m = frame ? MARK_KAITEKI2 : MARK_KAITEKI;
  }
  drawMarkS(m, 84, 4 + hop, MARK_SCALE);
}

void drawScreen(unsigned long now) {
  char line[32];
  display.clearBuffer();
  display.setFont(u8g2_font_b12_t_japanese2);
  if (mode == MODE_DICE) {
    drawDiceFace(rolling || hasResult ? shown : 1, 6, 8, 48);
    if (rolling) {
      display.drawUTF8(56, 62, "ごろごろ…");
    } else if (hasResult) {
      snprintf(line, sizeof(line), "%d が でた!", result);
      display.drawUTF8(56, 62, line);
    } else {
      display.drawUTF8(56, 62, "さわって!");
    }
  } else {
    display.drawUTF8(4, 20, "きょうは…");
    if (rolling) display.drawUTF8(4, 44, OMIKUJI[shown]);
    else if (hasResult) display.drawUTF8(4, 44, OMIKUJI[result]);
    else display.drawUTF8(4, 44, "？");
    display.drawUTF8(56, 62, rolling ? "ごろごろ…" : "おみくじ");
  }
  drawCactus(now);
  display.sendBuffer();
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  display.begin();
  display.setBusClock(400000);
  display.enableUTF8Print();
  display.setFont(u8g2_font_b12_t_japanese2);

  display.clearBuffer();
  display.drawUTF8(0, 30, "さいころ");
  drawMarkS(MARK_KAITEKI, 84, 4, MARK_SCALE);
  display.drawUTF8(0, 62, "さわらないでね");
  display.sendBuffer();

  pad1.base = measureBase(pad1.pin);
  Serial.printf("タッチの きじゅん=%.0f\n", pad1.base);
  randomSeed(esp_random() ^ micros());
}

void loop() {
  unsigned long now = millis();
  padUpdate(pad1, now);

  if (pad1.longEdge) {
    mode = (mode == MODE_DICE) ? MODE_OMIKUJI : MODE_DICE;
    rolling = false;
    hasResult = false;
  } else if (pad1.releasedEdge && pad1.heldMs < LONG_MS && !rolling) {
    // ちょん（はなした とき）で ふる
    rolling = true;
    hasResult = false;
    rollStart = now;
    nextChange = now;
  }

  if (rolling) {
    if (now >= nextChange) {
      shown = pickValue();
      float progress = (float)(now - rollStart) / ROLL_MS;
      if (progress > 1) progress = 1;
      nextChange = now + 60 + (unsigned long)(progress * 220);  // だんだん ゆっくりに
    }
    if (now - rollStart >= ROLL_MS) {
      rolling = false;
      hasResult = true;
      result = pickValue();
      shown = result;
      Serial.printf("けっか=%d\n", result);
    }
  }

  drawScreen(now);
  delay(30);
}

2. はんのうゲーム

さぼてんきくんが「！」になったら、すぐに、ワイヤーをさわります。はやさを、きそいます。

  1. ワイヤー1を、ちょんとさわって、スタート。「まってね…」と出ます。
  2. 2〜5秒あとに、画面が、白くなって、「いま!」と出ます。
  3. そのしゅんかんに、ワイヤーを、ちょん！ はやさ（びょう）が、出ます。
  4. 「いま!」の前にさわると、フライングです。
  • ワイヤー1が「1P」、ワイヤー2が「2P」です。先にさわった人の、はやさが出ます。
  • いちばんはやい記録は、ボードの中に、のこります。電源を切っても、のこります。
  • けっかの画面で、ちょんとさわると、もう一度、遊べます。ワイヤー1を、ながくさわると、はじめの画面にもどります。

かえてみよう

  • WAIT_MIN_MS と WAIT_MAX_MS：「いま!」までの、時間のはば。
  • RATE_GREAT など：「すごい!」「はやい!」になる、はやさの線。
  • WORD_GREAT など：画面に出る言葉。

//   ESP32 3V3     → OLED VDD

U8G2_SSD1306_128X64_NONAME_F_HW_I2C display(U8G2_R0, U8X8_PIN_NONE);
Preferences prefs;

const int MARK_SIZE = 16;
const int MARK_SCALE = 2;

struct TouchPad {
  int pin;
  float base;               // さわっていない時の 値（きじゅん）
  bool down;                // いま さわっているか
  int cnt;                  // ちょうし を みる ための かぞえ
  unsigned long pendMs;     // さわり はじめた とき
  unsigned long downMs;     // さわった ときこく
  bool longFired;
  bool pressedEdge;         // いま さわった
  bool releasedEdge;        // いま はなした
  bool longEdge;            // いま ながく さわった
  unsigned long heldMs;     // はなした とき の、さわっていた 長さ
};

// ESP32 は、さわると 値が 小さくなる
float measureBase(int pin) {
  long sum = 0;
  for (int i = 0; i < 16; i++) {
    sum += touchRead(pin);
    delay(4);
  }
  return sum / 16.0f;
}

void padUpdate(TouchPad &p, unsigned long now) {
  p.pressedEdge = false;
  p.releasedEdge = false;
  p.longEdge = false;
  float v = (float)touchRead(p.pin);
  bool raw = v < p.base * TOUCH_RATIO;
  if (raw == p.down) {
    p.cnt = 0;
    // さわっていない間は、きじゅんを、ゆっくり なおす
    if (!raw && v > p.base * 0.8f) p.base = p.base * 0.98f + v * 0.02f;
  } else {
    if (p.cnt == 0) p.pendMs = now;
    p.cnt++;
    if (p.cnt >= TOUCH_CONFIRM) {  // 3回つづけて 同じ → きまり（ちらつき よけ。かず は TOUCH_CONFIRM）
      p.down = raw;
      p.cnt = 0;
      if (raw) {
        p.pressedEdge = true;
        p.downMs = p.pendMs;
        p.longFired = false;
      } else {
        p.releasedEdge = true;
        p.heldMs = now - p.downMs;
      }
    }
  }
  if (p.down && !p.longFired && now - p.downMs >= LONG_MS) {
    p.longFired = true;
    p.longEdge = true;
  }
}

TouchPad pad1 = {14, 0, false, 0, 0, 0, false, false, false, false, 0};
TouchPad pad2 = {27, 0, false, 0, 0, 0, false, false, false, false, 0};

// ---- 0と1の絵を、画面にかく（scale は 1ドットを 何倍で出すか）----
void drawMarkS(const char *const mark[], int x, int y, int scale) {
  for (int row = 0; row < MARK_SIZE; row++) {
    const char *line = mark[row];
    for (int col = 0; col < MARK_SIZE && line[col] != '\0'; col++) {
      if (line[col] == '1') {
        display.drawBox(x + col * scale, y + row * scale, scale, scale);
      }
    }
  }
}
void drawMark(const char *const mark[], int x, int y) {
  drawMarkS(mark, x, y, MARK_SCALE);
}

const int S_IDLE = 0;    // はじめの画面
const int S_WAIT = 1;    // まってね…
const int S_GO = 2;      // いま！
const int S_RESULT = 3;  // けっか
int state = S_IDLE;

unsigned long stateMs = 0;    // この状態に なった とき
unsigned long waitMs = 0;     // 「いま！」までの 時間
unsigned long goMs = 0;       // 「いま！」を 出した とき
unsigned long bestMs = 99999; // いちばん はやい きろく
bool fly = false;             // フライング だったか
int who = 0;                  // さわった人（1 か 2）
unsigned long reactMs = 0;    // はんのう 時間
bool newBest = false;
bool needDraw = true;

void drawMarkFrame(const char *const a[], const char *const b[], int x, int y, int scale, unsigned long now) {
  int frame = (now / ANIME_MS) % 2;
  drawMarkS(frame ? b : a, x, y, scale);
}

void formatSec(char *out, size_t n, unsigned long ms) {
  snprintf(out, n, "%.2fびょう", ms / 1000.0f);
}

void drawScreen(unsigned long now) {
  char line[40];
  display.clearBuffer();
  display.setFont(u8g2_font_b12_t_japanese2);

  if (state == S_IDLE) {
    drawMarkFrame(MARK_KAITEKI, MARK_KAITEKI2, 96, 4, MARK_SCALE, now);
    display.drawUTF8(0, 13, "はんのうゲーム");
    display.drawUTF8(0, 29, "ちょんと");
    display.drawUTF8(0, 45, "スタート!");
    if (bestMs < 99999) {
      char t[24];
      formatSec(t, sizeof(t), bestMs);
      snprintf(line, sizeof(line), "ベスト %s", t);
      display.drawUTF8(0, 62, line);
    } else {
      display.drawUTF8(0, 62, "ベスト ---");
    }
  } else if (state == S_WAIT) {
    drawMarkS(MARK_KAITEKI, 48, 4, 3);
    display.drawUTF8(0, 62, "まってね…");
  } else if (state == S_GO) {
    // 画面が 白くなる（さぼてんきくんは びっくり）
    display.drawBox(0, 0, 128, 64);
    display.setDrawColor(0);
    drawMarkS(MARK_ITAI, 8, 8, 3);
    display.setFont(u8g2_font_logisoso28_tn);
    display.setDrawColor(0);
    display.drawStr(76, 40, "!");
    display.setFont(u8g2_font_b12_t_japanese2);
    display.drawUTF8(60, 62, "いま!");
    display.setDrawColor(1);
  } else {
    // けっか
    const char *const *m = fly ? MARK_ITAI : MARK_KAITEKI;
    const char *const *m2 = fly ? MARK_ITAI2 : MARK_KAITEKI2;
    const char *word = "";
    if (!fly) {
      if (reactMs < RATE_GREAT) { m = MARK_TERE; m2 = MARK_TERE2; word = WORD_GREAT; }
      else if (reactMs < RATE_GOOD) word = WORD_GOOD;
      else if (reactMs < RATE_NORMAL) word = WORD_NORMAL;
      else { m = MARK_KARAKARA; m2 = MARK_KARAKARA2; word = WORD_SLOW; }
    }
    drawMarkFrame(m, m2, 96, 4, MARK_SCALE, now);
    if (fly) {
      snprintf(line, sizeof(line), "%dP フライング!", who);
      display.drawUTF8(0, 13, line);
      display.drawUTF8(0, 29, "はやすぎ!");
    } else {
      char t[24];
      formatSec(t, sizeof(t), reactMs);
      snprintf(line, sizeof(line), "%dP", who);
      display.drawUTF8(0, 13, line);
      display.drawUTF8(0, 29, t);
      display.drawUTF8(0, 45, word);
    }
    if (newBest) {
      display.drawUTF8(0, 62, "しんきろく!");
    } else if (bestMs < 99999) {
      char t[24];
      formatSec(t, sizeof(t), bestMs);
      snprintf(line, sizeof(line), "ベスト %s", t);
      display.drawUTF8(0, 62, line);
    }
  }
  display.sendBuffer();
}

void startWait(unsigned long now) {
  state = S_WAIT;
  stateMs = now;
  waitMs = WAIT_MIN_MS + random(0, (long)(WAIT_MAX_MS - WAIT_MIN_MS + 1));
  needDraw = true;
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  display.begin();
  display.setBusClock(400000);
  display.enableUTF8Print();
  display.setFont(u8g2_font_b12_t_japanese2);

  display.clearBuffer();
  display.drawUTF8(0, 30, "はんのうゲーム");
  drawMarkS(MARK_KAITEKI, 96, 4, MARK_SCALE);
  display.drawUTF8(0, 62, "さわらないでね");
  display.sendBuffer();

  pad1.base = measureBase(pad1.pin);
  pad2.base = measureBase(pad2.pin);
  Serial.printf("タッチの きじゅん 1=%.0f 2=%.0f\n", pad1.base, pad2.base);

  prefs.begin("hannou", false);
  bestMs = prefs.getULong("best", 99999);
  randomSeed(esp_random() ^ micros());
}

void loop() {
  unsigned long now = millis();
  padUpdate(pad1, now);
  padUpdate(pad2, now);
  bool p1 = pad1.pressedEdge;
  bool p2 = pad2.pressedEdge;

  if (state == S_IDLE) {
    if (p1 || p2) startWait(now);
  } else if (state == S_WAIT) {
    if (p1 || p2) {
      // 「いま！」の まえに さわった → フライング
      fly = true;
      who = p1 ? 1 : 2;
      newBest = false;
      state = S_RESULT;
      stateMs = now;
      needDraw = true;
    } else if (now - stateMs >= waitMs) {
      state = S_GO;
      drawScreen(now);   // 先に 画面を 出してから、時間を はかりはじめる
      goMs = millis();
      stateMs = goMs;
    }
  } else if (state == S_GO) {
    if (p1 || p2) {
      who = p1 ? 1 : 2;
      unsigned long t = (who == 1) ? pad1.downMs : pad2.downMs;
      reactMs = (t > goMs) ? (t - goMs) : 0;
      fly = false;
      newBest = false;
      if (reactMs > 0 && reactMs < bestMs) {
        bestMs = reactMs;
        prefs.putULong("best", bestMs);
        newBest = true;
      }
      Serial.printf("%dP %lu ミリびょう\n", who, reactMs);
      state = S_RESULT;
      stateMs = now;
      needDraw = true;
    } else if (now - stateMs >= 5000) {
      // 5びょう たっても さわらない → もどる
      state = S_IDLE;
      needDraw = true;
    }
  } else {  // S_RESULT
    if ((p1 || p2) && now - stateMs >= 800) startWait(now);   // すぐ もういちど
    if (pad1.longEdge) { state = S_IDLE; needDraw = true; }    // ながく さわると はじめの画面へ
  }

  if (state != S_WAIT && state != S_GO) needDraw = true;       // 絵を うごかすため
  if (state == S_WAIT) needDraw = true;
  if (needDraw && state != S_GO) {
    drawScreen(now);
    needDraw = false;
  }
  delay(8);
}

  1. Arduino IDE で、「ファイル」→「新規スケッチ」を選び、はじめから書いてある文字を、ぜんぶ消す。
  2. 上の「…をコピーする」を押して、貼りつける。
