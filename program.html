プログラム（コピーして貼るだけ）

プログラムを、ボタンでコピーして、Arduino IDE に貼りつけるだけです。GitHub（ギットハブ）は、使いません。必要なのは、パソコンの準備（Arduino IDE・ボード・ライブラリ）だけです。

書き込みのやりかた

  1. ボードを、USBケーブルで、パソコンにつなぐ。
  2. Arduino IDE で、「ファイル」→「新規スケッチ」を選ぶ。新しい画面が開きます。
  3. はじめから書いてある文字（void setup() { … } と void loop() { … }）を、ぜんぶ消す。（画面の中をクリックして、Ctrl+A、Delete。Macは ⌘+A）
  4. 下の「プログラムをコピーする」を押して、画面の中に、貼りつける。（Ctrl+V。Macは ⌘+V）
  5. 「ツール」の「ボード」が「ESP32 Dev Module」、「ポート」が選ばれていることを確かめる。
  6. 左上の →（書き込み）ボタンを押す。「保存しますか？」と聞かれたら、名前を「sabotenkikei」にして、保存する。
  7. 下の黒い画面に「Connecting……」と出たまま進まないときは、ボードの BOOT ボタンを、「Writing」と出るまで押し続ける。
  8. 「Hard resetting」などと出て止まったら完了。数秒で、画面が切りかわります。

まちがえても、こわれません。わからなくなったら、このページから、もう一度コピーして、貼りつければ、出荷したときと同じ表示にもどります。

書きかえたものを残したいときは、「ファイル」→「名前を付けて保存」で、別の名前にしておくと安心です。

さぼてんきけい（温度・湿度・気圧）

いつもの、「さぼてんきけい」のプログラムです。「かえてみよう！」と書いてある場所を書きかえると、表示がかわります。（くわしくは、説明書のステップ2）

// さぼてんきけい（ESP32 + AHT20/BMP280 + 0.96インチOLED）
//
// このプログラムは Wi-Fi / Bluetooth を一切使いません（WiFi.h などを読み込んでいません）。
//
// 必要なライブラリ（Arduino IDE の「ライブラリを管理」から入れる）:
//   1. U8g2                (olikraus)
//   2. Adafruit BMP280 Library  ※「依存ライブラリも入れますか？」→「すべてインストール」
//
// ボード設定: ツール → ボード → esp32 → 「ESP32 Dev Module」
//
// 配線（I2C）:
//   ESP32 3V3     → センサー VDD / OLED VDD
//   ESP32 GND     → センサー GND / OLED GND
//   ESP32 GPIO21  → センサー SDA / OLED SDA
//   ESP32 GPIO22  → センサー SCL / OLED SCK
//   （基板の文字は、センサーが「VDD・SDA・GND・SCL」、OLEDが「GND・VDD・SCK・SDA」）
//
// 画面のようす（128 x 64 ドット）:
//   左がわ … 上から「温度」「湿度」「気圧」、いちばん下に「かいてき」などの言葉
//   右上   … しつどに合わせた、サボテンのキャラクター「さぼてんきくん」の顔（にこにこ・じめじめ・からから）。
//            顔は2まいの絵がこうたいに出て、動いて見えます。

#include <Wire.h>
#include <U8g2lib.h>
#include <Adafruit_BMP280.h>

// =====================================================================
//  かえてみよう！（ここから下を書きかえると、画面の表示がかわります）
//  書きかえたら、もう一度「書き込み」ボタン（→）をおしてね。
// =====================================================================

// ---- ① 数字をかえてみよう ----
// しつど（%）が この数字 より大きいと「じめじめ」、小さいと「からから」になります。
const float JIMEJIME_LINE = 70;  // これ以上で じめじめ
const float KARAKARA_LINE = 40;  // これより下で からから

// さぼてんきくんは、2まいの絵がこうたいに出て、動いているように見えます。
// その切りかえの間かくです。（ミリ秒。1000 = 1秒）小さいほど、はやく動きます。
// 50より小さくしないでね。
const unsigned long ANIME_MS = 500;

// ---- ② 言葉をかえてみよう ----
// " " の中を、すきな言葉にかえられます。（ひらがな・カタカナがおすすめ。
// むずかしい漢字は、画面に出ないことがあります）
const char *WORD_KAITEKI  = "かいてき";
const char *WORD_JIMEJIME = "じめじめ";
const char *WORD_KARAKARA = "からから";

// ---- ③ 0と1で 絵をかえてみよう ----

// =====================================================================
//  かえてみよう ここまで（ここから下は、なれてきたら読んでみよう）
// =====================================================================

// 0.96インチ OLED（SSD1306, 128x64, I2Cアドレス 0x3C）
U8G2_SSD1306_128X64_NONAME_F_HW_I2C display(U8G2_R0, U8X8_PIN_NONE);
Adafruit_BMP280 bmp;

const uint8_t AHT20_ADDR = 0x38;
bool bmpOk = false;

// 最後にはかった値（loop で入れかえて、drawScreen で使う）
float temp = 0, humi = 0, pressure = 0;
bool ahtOk = false;
const unsigned long SENSOR_MS = 1000;  // センサーを読む間かく（ミリ秒）
unsigned long lastSensorMs = 0;
bool firstRead = true;
int lastFrame = -1;

const int MARK_SIZE  = 16;  // 絵のたて・よこのドット数
const int MARK_SCALE = 2;   // 1ドットを何倍で出すか（2 → 32x32ドット）
const int MARK_X = 128 - MARK_SIZE * MARK_SCALE;  // 右はしにぴったり
const int MARK_Y = 4;

// ---- 0と1の絵を、画面にかく ----
// 文字が '1' のところだけ四角をぬる。行が短くても、そこで止まるので安全。
void drawMark(const char *const mark[], int x, int y) {
  for (int row = 0; row < MARK_SIZE; row++) {
    const char *line = mark[row];
    for (int col = 0; col < MARK_SIZE && line[col] != '\0'; col++) {
      if (line[col] == '1') {
        display.drawBox(x + col * MARK_SCALE, y + row * MARK_SCALE, MARK_SCALE, MARK_SCALE);
      }
    }
  }
}

// ---- AHT20（温度・湿度）をライブラリなしで読む ----
void ahtInit() {
  delay(40);
  Wire.beginTransmission(AHT20_ADDR);
  Wire.write(0xBE);
  Wire.write(0x08);
  Wire.write(0x00);
  Wire.endTransmission();
  delay(10);
}

bool ahtRead(float &tempC, float &humi) {
  Wire.beginTransmission(AHT20_ADDR);
  Wire.write(0xAC);
  Wire.write(0x33);
  Wire.write(0x00);
  if (Wire.endTransmission() != 0) return false;  // つながっていない
  delay(80);
  if (Wire.requestFrom((uint8_t)AHT20_ADDR, (uint8_t)7) != 7) return false;
  uint8_t d[7];
  for (int i = 0; i < 7; i++) d[i] = Wire.read();
  if (d[0] & 0x80) return false;  // まだ測定中
  uint32_t rawH = ((uint32_t)d[1] << 12) | ((uint32_t)d[2] << 4) | (d[3] >> 4);
  uint32_t rawT = ((uint32_t)(d[3] & 0x0F) << 16) | ((uint32_t)d[4] << 8) | d[5];
  humi  = rawH * 100.0f / 1048576.0f;
  tempC = rawT * 200.0f / 1048576.0f - 50.0f;
  return true;
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);  // SDA=21, SCL=22

  display.begin();
  display.enableUTF8Print();
  // 日本語フォント（「湿」を含むのは japanese2 以上。japanese1 には入っていない）
  display.setFont(u8g2_font_b12_t_japanese2);

  // はじめの画面（2秒）
  display.clearBuffer();
  display.drawUTF8(0, 36, "さぼてんきけい");
  drawMark(MARK_KAITEKI, MARK_X, 16);
  display.sendBuffer();

  ahtInit();

  // 気圧センサー BMP280 のI2Cアドレスは 0x76 か 0x77（商品により違う）
  bmpOk = bmp.begin(0x76) || bmp.begin(0x77);
  if (!bmpOk) Serial.println("BMP280 が見つかりません（配線を確認）");

  delay(2000);
}

// ---- 画面に、ぜんぶをかく（frame は 0 か 1。絵1か絵2か） ----
void drawScreen(int frame) {
  char line[40];
  display.clearBuffer();
  display.setFont(u8g2_font_b12_t_japanese2);

  if (ahtOk) {
    snprintf(line, sizeof(line), "温度 %.1f C", temp);
    display.drawUTF8(0, 13, line);

    snprintf(line, sizeof(line), "湿度 %.0f%%", humi);
    display.drawUTF8(0, 29, line);

    // しつどで、言葉と絵をえらぶ
    const char *word;
    const char *const *mark;
    if (humi >= JIMEJIME_LINE) {
      word = WORD_JIMEJIME;
      mark = frame ? MARK_JIMEJIME2 : MARK_JIMEJIME;
    } else if (humi < KARAKARA_LINE) {
      word = WORD_KARAKARA;
      mark = frame ? MARK_KARAKARA2 : MARK_KARAKARA;
    } else {
      word = WORD_KAITEKI;
      mark = frame ? MARK_KAITEKI2 : MARK_KAITEKI;
    }
    display.drawUTF8(0, 62, word);
    drawMark(mark, MARK_X, MARK_Y);
  } else {
    display.drawUTF8(0, 13, "センサーが");
    display.drawUTF8(0, 29, "つながって");
    display.drawUTF8(0, 62, "いません");
  }

  if (bmpOk) {
    snprintf(line, sizeof(line), "気圧 %.0f hPa", pressure);
    display.drawUTF8(0, 45, line);
  } else {
    display.drawUTF8(0, 45, "気圧 ---");
  }

  display.sendBuffer();
}

void loop() {
  unsigned long now = millis();

  // センサーは、1秒（SENSOR_MS）ごとに読む
  bool sensorTime = firstRead || (now - lastSensorMs >= SENSOR_MS);
  if (sensorTime) {
    ahtOk = ahtRead(temp, humi);
    if (bmpOk) pressure = bmp.readPressure() / 100.0f;
    Serial.printf("温度=%.1f 湿度=%.0f\n", temp, humi);
    lastSensorMs = now;
    firstRead = false;
  }

  // 絵は、ANIME_MS ごとに、絵1と絵2を入れかえる
  unsigned long animeMs = (ANIME_MS < 50) ? 50 : ANIME_MS;
  int frame = (now / animeMs) % 2;

  // かわったときだけ、画面をかきなおす
  if (sensorTime || frame != lastFrame) {
    lastFrame = frame;
    drawScreen(frame);
  }
  delay(10);
}

うまくいかないとき

  • 赤い文字で「No such file」と出る：ライブラリ（U8g2 と Adafruit BMP280）が、入っていません。パソコンの準備の「3. ライブラリを入れる」を、見てください。
  • 貼りつけたら、日本語が「？」になった：ふつうは、なりません。もう一度、「ぜんぶ消して」から、貼りつけてください。
  • その他：おうちの人へ「うまくいかないとき」を、見てください。
